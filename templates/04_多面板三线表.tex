% 依赖：booktabs
% ==============================================================================
\begin{table}[htbp]
  \centering
  \caption{ {{ caption }} }
  \label{tab:{{ label }}}
  \small

  {% for panel in panels %}面板 {{ panel.name }}：{{ panel.title }}

  \begin{tabular}{ {{ panel.column_format }} }
    \toprule
    {% for col in panel.columns %}{{ col }}{% if not loop.last %} & {% endif %}{% endfor %}\\
    \midrule
    {% for row in panel.rows %}{% for cell in row %}{{ cell }}{% if not loop.last %} & {% endif %}{% endfor %}\\
    {% endfor %}
    \bottomrule
  \end{tabular}
  {% if not loop.last %}\vspace{0.5em}{% endif %}{% endfor %}
\end{table}
